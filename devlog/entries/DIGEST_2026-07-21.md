## [mobile] Phase 38 — central juice/animation layer

**What:** a new `lib/juice/` module gives ~33 previously-bespoke
animated components one shared set of Reanimated primitives — screen
shake, impact flash, a status-proc pulse, number pops, standard
enter/exit transitions, a static spent-die treatment, and a haptics
wrapper. Combat adopts it first: damage/DoT float numbers, the
player-damage shake+flash, a new status-land pulse on the player
medallion, and card drop-confirm/rejection feedback. An owner jot
(used dice have no spent-state indicator) rode the same phase —
spent dice now desaturate.
**Why:** every animation was a one-off before this; reduced-motion
handling and instant-mode escape hatches (for seeded e2e) now live in
one place instead of being re-solved per component.
```diff
+import { useEffect } from 'react';
+import { useAnimatedStyle, useSharedValue, withSequence, withTiming } from 'react-native-reanimated';
+import { useReducedMotion } from '@/hooks/useReducedMotion';
+import { resolveJuiceMode } from './instant';
+import { JUICE_TIMING } from './juice.timing';
+
+export function useJuicePulse(triggerKey: number, intensity = 1) {
+    const reducedMotion = useReducedMotion();
+    const scale = useSharedValue(1);
+    useEffect(() => {
+        if (triggerKey <= 0) return;
+        if (resolveJuiceMode({ reducedMotion }) === 'instant') { scale.value = 1; return; }
+        const peak = pulsePeakScale(intensity);
+        scale.value = withSequence(
+            withTiming(peak, { duration: JUICE_TIMING.pulse.inMs }),
+            withTiming(1, { duration: JUICE_TIMING.pulse.outMs }),
+        );
+    }, [triggerKey, intensity, reducedMotion]);
+    return useAnimatedStyle(() => ({ transform: [{ scale: scale.value }] }));
+}
```
